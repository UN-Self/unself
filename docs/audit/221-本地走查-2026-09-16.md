### E.2 `modules/chat/scripts/ws-client.mjs`（WS 双端取证客户端）

```js
#!/usr/bin/env node
// SPDX-License-Identifier: AGPL-3.0-only
// 用法：node scripts/ws-client.mjs <token> <listen|read|wsframe> [messageIdsCsv|等待秒数] [sendText]
const PORT = process.env.WS_PORT || 8791;
const [token, mode, arg3, sendText] = process.argv.slice(2);
const base = `http://127.0.0.1:${PORT}`;
async function jsonFetch(path, init) {
  const res = await fetch(`${base}${path}`, init);
  const text = await res.text();
  let body; try { body = JSON.parse(text); } catch { body = text; }
  return { status: res.status, body };
}
const ws = new WebSocket(`ws://127.0.0.1:${PORT}/api/ws/public/1?token=${encodeURIComponent(token)}`);
const closeInfo = new Promise((r) => ws.addEventListener('close', (e) => r({ code: e.code, reason: e.reason })));
await new Promise((res, rej) => { ws.addEventListener('open', res); ws.addEventListener('error', () => rej(new Error('WS error'))); setTimeout(() => rej(new Error('WS timeout')), 8000); });
ws.addEventListener('message', (e) => {
  let f = e.data; try { f = JSON.parse(e.data); } catch {}
  if (f?.type === 'read_receipts') console.log(`WS_FRAME type=read_receipts messageId=${f.messageId} readBy=${f.userId} readAt=${f.readAt} batch=${(f.messageIds||[]).length}`);
  else if (f?.type === 'message') console.log(`WS_FRAME type=message id=${f.message?.id} sender=${f.message?.sender?.displayName} content=${String(f.message?.content).slice(0,40)}`);
  else console.log(`WS_FRAME ${JSON.stringify(f).slice(0, 160)}`);
});
if (mode === 'wsframe') {                      // 停用/过期取证：等待后发业务帧 → 期望 1008
  const waitSec = Number(arg3 || '5');
  await new Promise((r) => setTimeout(r, waitSec * 1000));
  ws.send(JSON.stringify({ type: 'send', content: `#221 停用后仍尝试发送（${Date.now()}）` }));
  await new Promise((r) => setTimeout(r, 2500));
} else if (mode === 'read') {                  // 拉分页 + 上报已读（+ 可选发消息）
  const page = await jsonFetch('/api/messages?kind=public&roomId=1&limit=20', { headers: { authorization: `Bearer ${token}` } });
  const msgs = page.body?.messages || [];
  console.log(`HTTP_GET /api/messages status=${page.status} count=${msgs.length}`);
  for (const m of msgs) console.log(`  msg=${m.id} from=${m.sender?.displayName} receipts=${JSON.stringify(m.readReceipts)}`);
  const ids = (arg3 ? arg3.split(',').map(Number) : msgs.map((m) => m.id)).filter(Boolean);
  if (ids.length) { const rep = await jsonFetch('/api/messages/read', { method: 'POST', headers: { 'content-type': 'application/json', authorization: `Bearer ${token}` }, body: JSON.stringify({ kind: 'public', roomId: 1, messageIds: ids }) });
    console.log(`HTTP_POST /api/messages/read status=${rep.status} body=${JSON.stringify(rep.body).slice(0,120)} ids=[${ids.join(',')}]`); }
  if (sendText) { const s = await jsonFetch('/api/v1/rooms/public/1/messages', { method: 'POST', headers: { 'content-type': 'application/json', authorization: `Bearer ${token}` }, body: JSON.stringify({ clientMessageId: crypto.randomUUID(), content: sendText, mentionUserIds: [], attachment: null }) });
    console.log(`HTTP_POST /api/v1/rooms/public/1/messages status=${s.status} id=${s.body?.message?.id} content=${s.body?.message?.content}`); }
  await new Promise((r) => setTimeout(r, 3000));
} else { await new Promise((r) => setTimeout(r, 30000)); }   // listen：收帧 30s
ws.close(1000, 'done');
const closed = await Promise.race([closeInfo, new Promise((r) => setTimeout(() => r(null), 1500))]);
console.log(closed ? `WS_CLOSE code=${closed.code} reason=${closed.reason}` : 'WS_CLOSE local');
```

import { SignJWT, importJWK, calculateJwkThumbprint } from '<repo>/modules/chat/node_modules/jose/dist/webapi/index.js';
import { readFile } from 'node:fs/promises';
const store = JSON.parse(await readFile('<repo>/modules/chat/scripts/.mint-token-keys.json', 'utf8'));
const kid = await calculateJwkThumbprint(store.publicJwk);
const key = await importJWK(store.privateJwk, 'ES256');
const token = await new SignJWT({ iss: 'unself-core', sub: process.argv[3] || '3', aud: 'chat', name: '走查丙' })
  .setProtectedHeader({ alg: 'ES256', kid }).setIssuedAt().setExpirationTime(`${process.argv[2] || 8}s`).sign(key);
console.log(token);
